import { useState, useCallback, useEffect } from 'react';
import { Glyph, getRandomGlyphs } from './data/glyphs';
import { AudienceMember, getRandomAudience, calculateJokeScore } from './data/audiences';
import { GlyphDeck, GlyphCard } from './components/GlyphCard';
import { AudiencePanel } from './components/AudienceCreature';
import { ScoreCalculator } from './components/ScoreCalculator';
import { cn } from './utils/cn';

type GameState = 'menu' | 'tutorial' | 'playing' | 'delivering' | 'result';

interface GameStats {
  influence: number;
  totalJokes: number;
  successfulJokes: number;
  consecutiveFailures: number;
  highScore: number;
}

interface TutorialStep {
  title: string;
  instruction: string;
  highlight?: string;
  action?: 'select' | 'deliver' | 'next';
}

const TUTORIAL_GLYPHS: Glyph[] = [
  { id: 'tutorial-1', symbol: '⧖', name: 'Temporal Displacement', category: 'timing', description: 'Perfect timing, or the absence thereof', color: 'blue' },
  { id: 'tutorial-2', symbol: '⧙', name: 'Recognition Flip', category: 'recognition', description: 'Seeing familiar things anew', color: 'purple' },
  { id: 'tutorial-3', symbol: '⧛', name: 'Self Deprecation', category: 'self-deprecation', description: 'Laughing at your own expense', color: 'green' },
  { id: 'tutorial-4', symbol: '⧗', name: 'Social Violation', category: 'violation', description: 'Breaking the unspoken rules', color: 'red' },
  { id: 'tutorial-5', symbol: '⏸', name: 'The Pregnant Pause', category: 'timing', description: 'Suspense that grows teeth', color: 'blue' },
  { id: 'tutorial-6', symbol: '↻', name: 'Recontextualization', category: 'recognition', description: 'Moving the goalposts of meaning', color: 'purple' },
  { id: 'tutorial-7', symbol: 'SAD', name: 'Melancholy Rain', category: 'emotion', description: 'Sadness with a smile', color: 'purple' },
  { id: 'tutorial-8', symbol: 'ABS', name: 'Pure Absurdity', category: 'cosmic', description: 'Logic need not apply', color: 'green' },
];

const TUTORIAL_AUDIENCE: AudienceMember = {
  id: 'tutorial-glimmerbeast',
  name: 'The Glimmerbeast',
  creature: 'A crystalline entity that hums with judgmental light',
  emoji: '💎',
  sensitivityGlyphs: ['body', 'violation', 'timing'],
  hates: ['body', 'violation'],
  loves: ['recognition', 'timing', 'self-deprecation'],
  reactionPattern: 'delight',
  influenceReward: 50,
  reactionWindow: 2000,
  description: 'Hates bodily humor. Loves clever misdirection.',
};

const TUTORIAL_STEPS: TutorialStep[] = [
  {
    title: 'Welcome to the Stage!',
    instruction: 'You are a comedian in a world where humor is expressed through mystical glyphs. Your goal: make the audience laugh by combining glyphs into jokes!',
    action: 'next',
  },
  {
    title: 'Your Hand',
    instruction: 'These are the glyphs in your hand. Click on them to select them for your joke. Try selecting the blue ⧖ (Temporal Displacement) and purple ⧙ (Recognition Flip).',
    highlight: 'select',
  },
  {
    title: 'Building Your Joke',
    instruction: 'Great! Now add the green ⧛ (Self Deprecation) to complete a perfect setup-twist-release structure!',
    highlight: 'select',
  },
  {
    title: 'The Audience',
    instruction: 'This is The Glimmerbeast - a crystalline judge. It HATES red glyphs (violation) but LOVES blue (timing) and purple (recognition). Watch its reactions!',
    action: 'next',
  },
  {
    title: 'Deliver Your Joke',
    instruction: 'You\'ve selected 3 glyphs! Click "Deliver Joke" to see how the audience reacts. Watch the score calculation - it\'s a beautiful thing.',
    highlight: 'deliver',
  },
  {
    title: 'Score Breakdown',
    instruction: 'Each glyph contributes to Delight, Confusion, and Disgust! Synergies like Timing + Self-Deprecation create "Awkward Charm" bonuses!',
    action: 'next',
  },
  {
    title: 'Ready for the Real Show?',
    instruction: 'You\'ve learned the basics! In the real game, you\'ll face multiple audiences with hidden preferences. Gain Influence to unlock new glyph dialects!',
    action: 'next',
  },
];

export function App() {
  const [gameState, setGameState] = useState<GameState>('menu');
  const [hand, setHand] = useState<Glyph[]>([]);
  const [selectedGlyphs, setSelectedGlyphs] = useState<Glyph[]>([]);
  const [audiences, setAudiences] = useState<AudienceMember[]>([]);
  const [reactions, setReactions] = useState<Record<string, { delight: number; disgust: number; confusion: number }>>({});
  const [stats, setStats] = useState<GameStats>({
    influence: 0,
    totalJokes: 0,
    successfulJokes: 0,
    consecutiveFailures: 0,
    highScore: 0,
  });
  const [lastJokeScore, setLastJokeScore] = useState<{ total: number; details: string } | null>(null);
  const [showHiddenSensitivity, setShowHiddenSensitivity] = useState(false);
  const [roundTime, setRoundTime] = useState(60);
  
  // Tutorial state
  const [tutorialStep, setTutorialStep] = useState(0);
  const [isFirstRound, setIsFirstRound] = useState(true);
  const [tutorialHand, setTutorialHand] = useState<Glyph[]>(TUTORIAL_GLYPHS);
  const [tutorialSelected, setTutorialSelected] = useState<Glyph[]>([]);
  const [tutorialAudience] = useState<AudienceMember>(TUTORIAL_AUDIENCE);
  const [showScoreCalc, setShowScoreCalc] = useState(false);

  const startGame = useCallback(() => {
    setGameState('tutorial');
    setTutorialStep(0);
    setTutorialHand(TUTORIAL_GLYPHS);
    setTutorialSelected([]);
    setHand(TUTORIAL_GLYPHS);
    setAudiences([TUTORIAL_AUDIENCE]);
    setSelectedGlyphs([]);
    setReactions({});
    setStats({ influence: 0, totalJokes: 0, successfulJokes: 0, consecutiveFailures: 0, highScore: 0 });
    setShowHiddenSensitivity(false);
    setRoundTime(60);
    setIsFirstRound(true);
  }, []);

  const selectGlyph = useCallback((glyph: Glyph) => {
    if (gameState === 'tutorial') {
      setTutorialSelected(prev => {
        const isSelected = prev.some(g => g.id === glyph.id);
        if (isSelected) {
          return prev.filter(g => g.id !== glyph.id);
        } else if (prev.length < 5) {
          return [...prev, glyph];
        }
        return prev;
      });
      setSelectedGlyphs(prev => {
        const isSelected = prev.some(g => g.id === glyph.id);
        if (isSelected) {
          return prev.filter(g => g.id !== glyph.id);
        } else if (prev.length < 5) {
          return [...prev, glyph];
        }
        return prev;
      });
    } else if (gameState === 'playing') {
      setSelectedGlyphs(prev => {
        const isSelected = prev.some(g => g.id === glyph.id && g.symbol === glyph.symbol);
        if (isSelected) {
          return prev.filter((_, i) => i !== prev.findIndex(pg => pg.id === glyph.id && pg.symbol === glyph.symbol));
        } else if (prev.length < 5) {
          return [...prev, glyph];
        }
        return prev;
      });
    }
  }, [gameState]);

  const deliverJoke = useCallback(() => {
    if (selectedGlyphs.length === 0) return;
    
    setGameState('delivering');
    setShowScoreCalc(true);
    
    const newReactions: Record<string, { delight: number; disgust: number; confusion: number }> = {};
    
    audiences.forEach(audience => {
      const score = calculateJokeScore(selectedGlyphs, audience);
      newReactions[audience.id] = {
        delight: score.delight,
        disgust: score.disgust,
        confusion: score.confusion,
      };
    });
    
    setReactions(newReactions);
  }, [selectedGlyphs, audiences]);

  const handleScoreComplete = useCallback((score: { delight: number; disgust: number; confusion: number; total: number }) => {
    setShowScoreCalc(false);
    
    const totalScore = score.total;
    
    let feedback = '';
    if (totalScore > 100) feedback = 'LEGENDARY! The audience is transcendent!';
    else if (totalScore > 50) feedback = 'Brilliant! They love it!';
    else if (totalScore > 0) feedback = 'Nice! A solid laugh!';
    else if (totalScore > -50) feedback = 'Meh. Some smiles, mostly confusion.';
    else if (totalScore > -100) feedback = 'Oof. They\'re not feeling it.';
    else feedback = 'CATASTROPHIC! The audience has left!';
    
    setLastJokeScore({
      total: totalScore,
      details: feedback,
    });
    
    const isSuccessful = totalScore > 0;
    setStats(prev => {
      const newConsecutive = isSuccessful ? 0 : prev.consecutiveFailures + 1;
      const newInfluence = prev.influence + (isSuccessful ? Math.round(totalScore / 10) : -10);
      
      return {
        ...prev,
        influence: Math.max(0, newInfluence),
        totalJokes: prev.totalJokes + 1,
        successfulJokes: prev.successfulJokes + (isSuccessful ? 1 : 0),
        consecutiveFailures: newConsecutive,
        highScore: Math.max(prev.highScore, prev.influence + (isSuccessful ? Math.round(totalScore / 10) : 0)),
      };
    });
    
    if (stats.consecutiveFailures >= 2) {
      setShowHiddenSensitivity(true);
    }
    
    setTimeout(() => {
      setGameState('result');
    }, 2000);
  }, [stats.consecutiveFailures]);

  const nextRound = useCallback(() => {
    if (isFirstRound) {
      setIsFirstRound(false);
      setHand(getRandomGlyphs(8));
      setAudiences(getRandomAudience(3));
      setTutorialHand([]);
      setTutorialSelected([]);
      setSelectedGlyphs([]);
      setReactions({});
      setLastJokeScore(null);
      setRoundTime(60);
      setGameState('playing');
    } else {
      setHand(getRandomGlyphs(8));
      setSelectedGlyphs([]);
      setReactions({});
      setLastJokeScore(null);
      setRoundTime(60);
      setGameState('playing');
    }
  }, [isFirstRound]);

  const advanceTutorial = useCallback(() => {
    if (tutorialStep < TUTORIAL_STEPS.length - 1) {
      setTutorialStep(prev => prev + 1);
    } else {
      setIsFirstRound(false);
      setHand(getRandomGlyphs(8));
      setAudiences(getRandomAudience(3));
      setTutorialHand([]);
      setTutorialSelected([]);
      setSelectedGlyphs([]);
      setReactions({});
      setLastJokeScore(null);
      setRoundTime(60);
      setGameState('playing');
    }
  }, [tutorialStep]);

  const tutorialSelectGlyph = useCallback((glyph: Glyph) => {
    setTutorialSelected(prev => {
      const isSelected = prev.some(g => g.id === glyph.id);
      if (isSelected) {
        return prev.filter(g => g.id !== glyph.id);
      } else if (prev.length < 5) {
        return [...prev, glyph];
      }
      return prev;
    });
    setSelectedGlyphs(prev => {
      const isSelected = prev.some(g => g.id === glyph.id);
      if (isSelected) {
        return prev.filter(g => g.id !== glyph.id);
      } else if (prev.length < 5) {
        return [...prev, glyph];
      }
      return prev;
    });
  }, []);

  const tutorialDeliverJoke = useCallback(() => {
    setGameState('delivering');
    setShowScoreCalc(true);
    
    const score = calculateJokeScore(tutorialSelected, tutorialAudience);
    setReactions({ [tutorialAudience.id]: { delight: score.delight, disgust: score.disgust, confusion: score.confusion } });
  }, [tutorialSelected, tutorialAudience]);

  const handleTutorialScoreComplete = useCallback((score: { delight: number; disgust: number; confusion: number; total: number }) => {
    setShowScoreCalc(false);
    setLastJokeScore({
      total: score.total,
      details: 'Tutorial complete! Ready for the real show?',
    });
    setTimeout(() => {
      setGameState('result');
    }, 1500);
  }, []);

  useEffect(() => {
    if (gameState !== 'playing' && gameState !== 'tutorial') return;
    
    const timer = setInterval(() => {
      setRoundTime(prev => {
        if (prev <= 1) {
          if (selectedGlyphs.length > 0) {
            deliverJoke();
          } else {
            setGameState('result');
          }
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
    
    return () => clearInterval(timer);
  }, [gameState, selectedGlyphs, deliverJoke]);

  useEffect(() => {
    if (gameState !== 'tutorial') return;
    
    const step = TUTORIAL_STEPS[tutorialStep];
    if (!step) return;
    
    if (step.action === 'next') {
      const timer = setTimeout(advanceTutorial, 500);
      return () => clearTimeout(timer);
    }
  }, [gameState, tutorialStep, advanceTutorial]);

  // Tutorial rendering
  if (gameState === 'tutorial') {
    const step = TUTORIAL_STEPS[tutorialStep];
    const currentHand = tutorialStep < 4 ? tutorialHand : hand;
    const currentSelected = tutorialStep < 4 ? tutorialSelected : selectedGlyphs;
    const currentAudiences = tutorialStep >= 3 ? [tutorialAudience] : audiences;
    const isDelivering = false;
    const isMenu = false;
    
    return (
      <div className="min-h-screen bg-gradient-to-br from-indigo-900 via-purple-900 to-black flex flex-col">
        <div className="p-4 bg-black/30 backdrop-blur-sm border-b border-white/10">
          <div className="max-w-6xl mx-auto flex items-center justify-between">
            <div className="text-2xl">📖 Tutorial</div>
            <div className="text-white font-bold">Step {tutorialStep + 1} of {TUTORIAL_STEPS.length}</div>
          </div>
        </div>

        <div className="flex-1 flex flex-col p-4">
          <div className="max-w-4xl mx-auto mb-6">
            <div className="bg-gradient-to-r from-purple-800/50 to-indigo-800/50 rounded-2xl p-6 border border-purple-500/30">
              <h2 className="text-2xl font-bold text-white mb-2">{step.title}</h2>
              <p className="text-gray-300">{step.instruction}</p>
            </div>
          </div>

          {tutorialStep >= 3 && (
            <div className="p-4">
              <AudiencePanel
                audiences={currentAudiences}
                reactions={reactions}
                isProcessing={isDelivering}
                showHiddenSensitivity={showHiddenSensitivity}
              />
            </div>
          )}

          <div className="p-4 bg-black/20">
            <div className="max-w-4xl mx-auto">
              <div className="flex items-center justify-between mb-2">
                <h3 className="text-white font-bold">Your Joke ({currentSelected.length}/5)</h3>
                {currentSelected.length > 0 && tutorialStep >= 4 && (
                  <button
                    onClick={tutorialDeliverJoke}
                    disabled={false}
                    className="px-6 py-2 bg-gradient-to-r from-green-500 to-emerald-600 text-white font-bold rounded-lg hover:from-green-600 hover:to-emerald-700 transition-all disabled:opacity-50"
                  >
                    🎤 Deliver Joke!
                  </button>
                )}
              </div>
              <div className="flex gap-2 min-h-[100px] items-center justify-center bg-black/30 rounded-xl p-4">
                {currentSelected.length === 0 ? (
                  <p className="text-gray-500 italic">Select glyphs from your hand to build your joke...</p>
                ) : (
                  <div className="flex gap-2 flex-wrap justify-center">
                    {currentSelected.map((glyph, index) => (
                      <div key={`${glyph.id}-${index}`} className="relative">
                        <GlyphCard glyph={glyph} size="lg" />
                      </div>
                    ))}
                  </div>
                )}
              </div>
            </div>
          </div>

          <div className="flex-1 p-4">
            <div className="max-w-4xl mx-auto">
              <h3 className="text-white font-bold mb-2">Your Hand</h3>
              <GlyphDeck
                glyphs={currentHand}
                onGlyphSelect={tutorialSelectGlyph}
                selectedGlyphs={currentSelected}
                disabled={isMenu}
              />
            </div>
          </div>
        </div>

        {showScoreCalc && currentSelected.length > 0 && (
          <ScoreCalculator
            jokeGlyphs={currentSelected}
            audience={tutorialAudience}
            onComplete={handleTutorialScoreComplete}
          />
        )}

        {/* Result overlay - handled by ScoreCalculator */}
        {false}
      </div>
    );
  }

  // Menu
  if (gameState === 'menu') {
    return (
      <div className="min-h-screen bg-gradient-to-br from-purple-900 via-indigo-900 to-black flex flex-col items-center justify-center p-8">
        <div className="text-center space-y-8 max-w-2xl">
          <div className="space-y-4">
            <h1 className="text-5xl md:text-7xl font-bold text-transparent bg-clip-text bg-gradient-to-r from-yellow-400 via-pink-500 to-purple-500">
              The Absurd Art
            </h1>
            <h2 className="text-3xl md:text-4xl font-bold text-white">
              of Making Them Laugh
            </h2>
            <p className="text-xl text-gray-300 italic">
              A game of glyphs, giggles, and glorious misinterpretation
            </p>
          </div>

          <div className="flex justify-center gap-4 text-4xl">
            <span className="animate-bounce" style={{ animationDelay: '0ms' }}>⧖</span>
            <span className="animate-bounce" style={{ animationDelay: '100ms' }}>⧗</span>
            <span className="animate-bounce" style={{ animationDelay: '200ms' }}>⧙</span>
            <span className="animate-bounce" style={{ animationDelay: '300ms' }}>⧛</span>
            <span className="animate-bounce" style={{ animationDelay: '400ms' }}>🜚</span>
          </div>

          <div className="space-y-4">
            <button
              onClick={startGame}
              className="w-full max-w-md px-8 py-4 bg-gradient-to-r from-pink-500 to-purple-600 text-white text-xl font-bold rounded-xl hover:from-pink-600 hover:to-purple-700 transition-all transform hover:scale-105 shadow-lg shadow-purple-500/50"
            >
              🎭 Start Performance
            </button>
            
            {stats.highScore > 0 && (
              <div className="text-gray-400">
                High Score: <span className="text-yellow-400 font-bold">{stats.highScore}</span> Influence
              </div>
            )}
          </div>

          <div className="bg-black/30 rounded-2xl p-6 text-left space-y-4 backdrop-blur-sm">
            <h3 className="text-xl font-bold text-white text-center">How to Play</h3>
            <div className="grid md:grid-cols-3 gap-4 text-sm text-gray-300">
              <div className="space-y-2">
                <div className="text-2xl">🎴</div>
                <p><strong className="text-white">Select Glyphs</strong></p>
                <p>Choose 3-5 symbol cards to craft your joke</p>
              </div>
              <div className="space-y-2">
                <div className="text-2xl">👁️</div>
                <p><strong className="text-white">Read the Audience</strong></p>
                <p>Each creature has hidden preferences</p>
              </div>
              <div className="space-y-2">
                <div className="text-2xl">🎤</div>
                <p><strong className="text-white">Deliver & Score</strong></p>
                <p>Watch reactions and gain Influence</p>
              </div>
            </div>
          </div>

          <div className="bg-black/30 rounded-2xl p-6 backdrop-blur-sm">
            <h3 className="text-lg font-bold text-white mb-4">Glyph Categories</h3>
            <div className="grid grid-cols-2 md:grid-cols-4 gap-3 text-xs">
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-blue-500"></span>
                <span className="text-gray-300">Timing & Structure</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-red-500"></span>
                <span className="text-gray-300">Violation & Risk</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-purple-500"></span>
                <span className="text-gray-300">Recognition & Flip</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-green-500"></span>
                <span className="text-gray-300">Cosmic & Surreal</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-amber-500"></span>
                <span className="text-gray-300">Body & Physical</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-cyan-500"></span>
                <span className="text-gray-300">Logic & Absurd</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-pink-500"></span>
                <span className="text-gray-300">Emotion & Melancholy</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="w-4 h-4 rounded bg-gray-500"></span>
                <span className="text-gray-300">Self-Deprecation</span>
              </div>
            </div>
          </div>
        </div>
      </div>
    );
  }

  // Playing / Delivering / Result
  const isDelivering = gameState === 'delivering';
  const isResult = gameState === 'result';
  const isMenu = false;
  
  return (
    <div className="min-h-screen bg-gradient-to-br from-indigo-900 via-purple-900 to-black flex flex-col">
      <div className="p-4 bg-black/30 backdrop-blur-sm border-b border-white/10">
        <div className="max-w-6xl mx-auto flex items-center justify-between">
          <div className="flex items-center gap-4">
            <div className="text-2xl">🎭</div>
            <div>
              <div className="text-white font-bold">Influence: <span className="text-yellow-400">{stats.influence}</span></div>
              <div className="text-xs text-gray-400">Jokes: {stats.successfulJokes}/{stats.totalJokes}</div>
            </div>
          </div>
          
          <div className="flex items-center gap-4">
            <div className={cn(
              'text-4xl font-bold text-white',
              roundTime <= 10 && 'text-red-400 animate-pulse'
            )}>
              {roundTime}s
            </div>
            <button
              onClick={() => setGameState('menu')}
              className="px-4 py-2 bg-gray-700 text-white rounded-lg hover:bg-gray-600 transition-colors"
            >
              Menu
            </button>
          </div>
        </div>
      </div>

      <div className="flex-1 flex flex-col">
        <div className="p-4">
          <AudiencePanel
            audiences={audiences}
            reactions={reactions}
            isProcessing={isDelivering}
            showHiddenSensitivity={showHiddenSensitivity}
          />
        </div>

        <div className="p-4 bg-black/20">
          <div className="max-w-4xl mx-auto">
            <div className="flex items-center justify-between mb-2">
              <h3 className="text-white font-bold">Your Joke ({selectedGlyphs.length}/5)</h3>
              {selectedGlyphs.length > 0 && (
                <button
                  onClick={deliverJoke}
                  disabled={isDelivering || isResult || isMenu}
                  className="px-6 py-2 bg-gradient-to-r from-green-500 to-emerald-600 text-white font-bold rounded-lg hover:from-green-600 hover:to-emerald-700 transition-all disabled:opacity-50"
                >
                  🎤 Deliver Joke!
                </button>
              )}
            </div>
            <div className="flex gap-2 min-h-[100px] items-center justify-center bg-black/30 rounded-xl p-4">
              {selectedGlyphs.length === 0 ? (
                <p className="text-gray-500 italic">Select glyphs from your hand to build your joke...</p>
              ) : (
                <div className="flex gap-2 flex-wrap justify-center">
                  {selectedGlyphs.map((glyph, index) => (
                    <div key={`${glyph.id}-${index}`} className="relative">
                      <GlyphCard glyph={glyph} size="lg" />
                      <button
                        onClick={() => setSelectedGlyphs(selectedGlyphs.filter((_, i) => i !== index))}
                        className="absolute -top-2 -right-2 w-6 h-6 bg-red-500 text-white rounded-full text-xs hover:bg-red-600"
                      >
                        ×
                      </button>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        </div>

        <div className="flex-1 p-4">
          <div className="max-w-4xl mx-auto">
            <h3 className="text-white font-bold mb-2">Your Hand</h3>
            <GlyphDeck
              glyphs={hand}
              onGlyphSelect={selectGlyph}
              selectedGlyphs={selectedGlyphs}
              disabled={isMenu}
            />
          </div>
        </div>
      </div>

      {showScoreCalc && selectedGlyphs.length > 0 && (
        <ScoreCalculator
          jokeGlyphs={selectedGlyphs}
          audience={audiences[0]}
          onComplete={handleScoreComplete}
        />
      )}

      {gameState === 'result' && lastJokeScore && (
        <div className="fixed inset-0 bg-black/80 flex items-center justify-center z-50">
          <div className="bg-gradient-to-br from-purple-800 to-indigo-900 rounded-2xl p-8 max-w-md text-center space-y-6 border border-purple-500/50 shadow-2xl">
            <div className="text-6xl">
              {lastJokeScore.total > 50 ? '🎉' : lastJokeScore.total > 0 ? '😊' : '😐'}
            </div>
            <div>
              <div className="text-3xl font-bold text-white mb-2">
                {Math.round(lastJokeScore.total)} Points
              </div>
              <p className="text-gray-300">{lastJokeScore.details}</p>
            </div>
            
            <div className="flex gap-2 justify-center text-sm">
              {Object.entries(reactions).map(([id, reaction]) => {
                const audience = audiences.find(a => a.id === id);
                return (
                  <div key={id} className="bg-black/30 rounded-lg px-3 py-2">
                    <div className="text-lg">{audience?.emoji}</div>
                    <div className="text-xs text-gray-400">
                      {reaction.delight > reaction.disgust ? '😄' : '😕'}
                    </div>
                  </div>
                );
              })}
            </div>

            <button
              onClick={nextRound}
              className="w-full px-6 py-3 bg-gradient-to-r from-pink-500 to-purple-600 text-white font-bold rounded-xl hover:from-pink-600 hover:to-purple-700 transition-all"
            >
              Next Round →
            </button>

            {stats.consecutiveFailures >= 3 && (
              <div className="text-yellow-400 text-sm animate-pulse">
                ✨ Self-Awareness Unlocked! Hidden preferences revealed!
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
}

export default App;
