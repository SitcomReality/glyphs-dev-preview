import React, { useEffect, useState } from 'react';
import { Glyph } from '../data/glyphs';
import { AudienceMember } from '../data/audiences';
import { calculateJokeScore } from '../data/audiences';
import { cn } from '../utils/cn';

interface FloatingScore {
  id: number;
  text: string;
  value: number;
  color: string;
  x: number;
  y: number;
}

interface ScoreCalculatorProps {
  jokeGlyphs: Glyph[];
  audience: AudienceMember;
  onComplete: (score: { delight: number; disgust: number; confusion: number; total: number }) => void;
}

export const ScoreCalculator: React.FC<ScoreCalculatorProps> = ({ jokeGlyphs, audience, onComplete }) => {
  const [floatingScores, setFloatingScores] = useState<FloatingScore[]>([]);
  const [currentIndex, setCurrentIndex] = useState(0);
  const [runningTotal, setRunningTotal] = useState({ delight: 0, disgust: 0, confusion: 0, total: 0 });
  const [isProcessing, setIsProcessing] = useState(true);

  useEffect(() => {
    if (currentIndex >= jokeGlyphs.length || !isProcessing) return;

    const glyph = jokeGlyphs[currentIndex];
    const score = calculateJokeScore([glyph], audience);
    
    // Calculate synergy with previously processed glyphs
    let synergyBonus = 0;
    const synergyTexts: string[] = [];
    
    jokeGlyphs.slice(0, currentIndex).forEach(prevGlyph => {
      // violation + recognition = dark humor
      if ((prevGlyph.category === 'violation' && glyph.category === 'recognition') ||
          (prevGlyph.category === 'recognition' && glyph.category === 'violation')) {
        synergyBonus += 15;
        synergyTexts.push('Dark Humor!');
      }
      // timing + self-deprecation = awkward charm
      if ((prevGlyph.category === 'timing' && glyph.category === 'self-deprecation') ||
          (glyph.category === 'timing' && prevGlyph.category === 'self-deprecation')) {
        synergyBonus += 10;
        synergyTexts.push('Awkward Charm!');
      }
      // body + emotion = melancholy gold
      if ((prevGlyph.category === 'body' && glyph.category === 'emotion') ||
          (glyph.category === 'body' && prevGlyph.category === 'emotion')) {
        synergyBonus += 12;
        synergyTexts.push('Melancholy Gold!');
      }
      // violation + violation = audience walks out
      if (prevGlyph.category === 'violation' && glyph.category === 'violation') {
        synergyBonus -= 30;
        synergyTexts.push('Audience Walks Out!');
      }
    });

    const totalValue = score.total + synergyBonus;
    
    const newFloating: FloatingScore = {
      id: Date.now() + currentIndex,
      text: `${glyph.name}${synergyTexts.length > 0 ? ` + ${synergyTexts.join(', ')}` : ''}`,
      value: totalValue,
      color: totalValue > 0 ? 'text-green-400' : totalValue < 0 ? 'text-red-400' : 'text-yellow-400',
      x: 50 + (Math.random() - 0.5) * 100,
      y: 50,
    };

    setFloatingScores(prev => [...prev, newFloating]);
    
    const newTotal = {
      delight: runningTotal.delight + score.delight,
      disgust: runningTotal.disgust + score.disgust,
      confusion: runningTotal.confusion + score.confusion,
      total: runningTotal.total + totalValue,
    };
    
    setRunningTotal(newTotal);
    setCurrentIndex(prev => prev + 1);

    // Remove floating score after animation
    setTimeout(() => {
      setFloatingScores(prev => prev.filter(f => f.id !== newFloating.id));
    }, 1500);

  }, [currentIndex, jokeGlyphs, audience, isProcessing, runningTotal]);

  useEffect(() => {
    if (currentIndex >= jokeGlyphs.length && isProcessing) {
      setIsProcessing(false);
      setTimeout(() => {
        onComplete(runningTotal);
      }, 1000);
    }
  }, [currentIndex, jokeGlyphs.length, isProcessing, onComplete, runningTotal]);

  const getScoreDescription = (total: number) => {
    if (total > 100) return { text: 'LEGENDARY!', color: 'text-yellow-300', emoji: '🌟' };
    if (total > 50) return { text: 'Brilliant!', color: 'text-green-300', emoji: '✨' };
    if (total > 0) return { text: 'Nice!', color: 'text-blue-300', emoji: '👍' };
    if (total > -50) return { text: 'Meh...', color: 'text-gray-300', emoji: '😐' };
    if (total > -100) return { text: 'Oof...', color: 'text-orange-300', emoji: '😅' };
    return { text: 'CATASTROPHIC!', color: 'text-red-300', emoji: '💀' };
  };

  const description = getScoreDescription(runningTotal.total);

  return (
    <div className="fixed inset-0 bg-black/90 flex items-center justify-center z-50 overflow-hidden">
      {/* Background effects */}
      <div className="absolute inset-0 bg-gradient-to-br from-purple-900/50 via-indigo-900/50 to-black/50" />
      
      {/* Animated background circles */}
      <div className="absolute inset-0 overflow-hidden">
        {[...Array(5)].map((_, i) => (
          <div
            key={i}
            className="absolute rounded-full border border-purple-500/20 animate-ping"
            style={{
              width: `${200 + i * 150}px`,
              height: `${200 + i * 150}px`,
              left: '50%',
              top: '50%',
              transform: 'translate(-50%, -50%)',
              animationDelay: `${i * 0.3}s`,
              animationDuration: '3s',
            }}
          />
        ))}
      </div>

      <div className="relative z-10 flex flex-col items-center gap-8 max-w-2xl w-full px-8">
        {/* Audience member being evaluated */}
        <div className="flex flex-col items-center gap-4">
          <div className="text-6xl animate-bounce-in">{audience.emoji}</div>
          <h2 className="text-2xl font-bold text-white">{audience.name}</h2>
          <p className="text-gray-400 text-center text-sm">{audience.description}</p>
        </div>

        {/* Processing area */}
        <div className="relative w-full h-64 flex items-center justify-center">
          {/* Floating scores */}
          {floatingScores.map(score => (
            <div
              key={score.id}
              className={cn(
                'floating-score text-2xl',
                score.color
              )}
              style={{
                left: `${score.x}%`,
                top: `${score.y}px`,
                animation: 'float-up 1.5s ease-out forwards',
              }}
            >
              {score.text} <span className={score.color}>{score.value > 0 ? '+' : ''}{score.value}</span>
            </div>
          ))}

          {/* Current glyph being processed */}
          {currentIndex < jokeGlyphs.length && isProcessing && (
            <div className="relative">
              <div className="w-24 h-32 rounded-xl bg-gradient-to-br from-purple-600 to-purple-900 border-2 border-purple-400 flex flex-col items-center justify-center shadow-2xl animate-pulse-glow">
                <span className="text-4xl text-white font-bold">{jokeGlyphs[currentIndex].symbol}</span>
                <span className="text-xs text-white/80 mt-2 text-center px-2">{jokeGlyphs[currentIndex].name}</span>
              </div>
              <div className="absolute -bottom-2 left-1/2 -translate-x-1/2 flex gap-1">
                <span className="w-2 h-2 bg-yellow-400 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
                <span className="w-2 h-2 bg-yellow-400 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
                <span className="w-2 h-2 bg-yellow-400 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
              </div>
            </div>
          )}

          {/* Final score reveal */}
          {!isProcessing && (
            <div className="text-center animate-bounce-in">
              <div className="text-6xl mb-4">{description.emoji}</div>
              <div className={cn('text-4xl font-bold', description.color)}>
                {runningTotal.total > 0 ? '+' : ''}{Math.round(runningTotal.total)}
              </div>
              <div className={cn('text-xl mt-2', description.color)}>{description.text}</div>
            </div>
          )}
        </div>

        {/* Running totals */}
        <div className="w-full grid grid-cols-4 gap-4">
          <div className="bg-black/50 rounded-xl p-4 text-center border border-green-500/30">
            <div className="text-green-400 text-sm">Delight</div>
            <div className="text-2xl font-bold text-green-300">{Math.round(runningTotal.delight)}</div>
          </div>
          <div className="bg-black/50 rounded-xl p-4 text-center border border-yellow-500/30">
            <div className="text-yellow-400 text-sm">Confusion</div>
            <div className="text-2xl font-bold text-yellow-300">{Math.round(runningTotal.confusion)}</div>
          </div>
          <div className="bg-black/50 rounded-xl p-4 text-center border border-red-500/30">
            <div className="text-red-400 text-sm">Disgust</div>
            <div className="text-2xl font-bold text-red-300">{Math.round(runningTotal.disgust)}</div>
          </div>
          <div className="bg-black/50 rounded-xl p-4 text-center border border-purple-500/30">
            <div className="text-purple-400 text-sm">Total</div>
            <div className={cn('text-2xl font-bold', runningTotal.total > 0 ? 'text-green-300' : 'text-red-300')}>
              {runningTotal.total > 0 ? '+' : ''}{Math.round(runningTotal.total)}
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};
