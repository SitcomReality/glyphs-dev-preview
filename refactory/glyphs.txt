export const getRandomGlyphs = (count: number): Glyph[] => {
  const shuffled = [...GLYPHS].sort(() => Math.random() - 0.5);
  return shuffled.slice(0, count);
};

export const getGlyphById = (id: string): Glyph | undefined => {
  return GLYPHS.find(g => g.id === id);
};
